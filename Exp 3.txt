#include <stdio.h>

int main()
{
    char s[100];
    int i;

    printf("Enter: ");
    fgets(s, 100, stdin);

    for(i=0; s[i]!='\0'; i++)
    {
        if(s[i]=='/' && s[i+1]=='/')
            break;

        if(s[i]!=' ' && s[i]!='\n' && s[i]!='\t')
            printf("%c", s[i]);
    }

    return 0;
}
